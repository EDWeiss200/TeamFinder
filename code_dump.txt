async def test_domain_exceptions_map_to_proper_http_status_codes(
    async_client: AsyncClient,
):
    """Проверяет, что exception_handlers.py возвращает 404, 403 и 409 для доменных исключений."""
    me = make_user()
    app.dependency_overrides[current_user] = lambda: me

    mock_service = AsyncMock()
    app.dependency_overrides[user_service] = lambda: mock_service

    # 1. EntityNotFoundError -> 404 Not Found
    mock_service.get_user_by_id.side_effect = EntityNotFoundError("Не найдено")
    res_404 = await async_client.get(f"/api/v1/users/{uuid.uuid4()}")
    assert res_404.status_code == 404

    # 2. PermissionDeniedError -> 403 Forbidden
    mock_service.get_user_by_id.side_effect = PermissionDeniedError("Нет прав")
    res_403 = await async_client.get(f"/api/v1/users/{uuid.uuid4()}")
    assert res_403.status_code == 403

    # 3. TeamIsFullError & AlreadyInTeamError -> 409 Conflict
    mock_service.get_user_by_id.side_effect = TeamIsFullError("Мест нет")
    res_409_full = await async_client.get(f"/api/v1/users/{uuid.uuid4()}")
    assert res_409_full.status_code == 409

    mock_service.get_user_by_id.side_effect = AlreadyInTeamError("Уже в команде")
    res_409_in_team = await async_client.get(f"/api/v1/users/{uuid.uuid4()}")
    assert res_409_in_team.status_code == 409

async def test_lazy_transition_registration_to_team_lock(fake_uow: FakeUnitOfWork):
    """Если registration_end в прошлом, статус переходит в TEAM_LOCK, а висящие заявки отменяются."""
    past = datetime.now(timezone.utc) - timedelta(hours=1)
    event = make_event(status=EventStatus.REGISTRATION, registration_end=past)
    fake_uow.events.get_by_id_or_invite_code.return_value = event

    service = EventServices(fake_uow)

    )
    assert fake_uow.committed is True

import io

import pytest
from fastapi import UploadFile

from services.upload_services import MAX_IMAGE_SIZE_BYTES, UploadServices

def make_upload_file(content: bytes, filename: str, content_type: str) -> UploadFile:
    """Вспомогательная фабрика UploadFile с заданным MIME-типом."""
    return UploadFile(
        file=io.BytesIO(content),
        filename=filename,
        headers=Headers({"content-type": content_type}),
    )


def generate_valid_image_bytes(fmt: str = "PNG") -> bytes:
    """Генерирует валидное изображение 10x10 пикселей в памяти через Pillow."""
    buf = io.BytesIO()
    img = Image.new("RGB", (10, 10), color="blue")
    img.save(buf, format=fmt)
    return buf.getvalue()


async def test_save_image_valid_png(tmp_path: Path):
    """Валидный PNG-файл успешно проходит проверку Pillow и сохраняется на диск."""
    service = UploadServices(upload_dir=tmp_path)
    img_bytes = generate_valid_image_bytes("PNG")
    upload = make_upload_file(img_bytes, "avatar.png", "image/png")

    url = await service.save_image(upload)

    assert url.startswith("/static/uploads/")
    assert url.endswith(".png")

from fastapi import FastAPI, Request, status
from fastapi.responses import JSONResponse

from utils.exceptions import (
    AlreadyInTeamError,
    DeadlinePassedError,
    DomainError,
    EntityNotFoundError,
    IncompleteTeamError,
    InvalidStateError,
    PermissionDeniedError,
    TeamIsFullError,
)


def register_exception_handlers(app: FastAPI) -> None:
    """Регистрация обработчиков исключений доменной логики в приложении FastAPI."""

    @app.exception_handler(EntityNotFoundError)
    async def entity_not_found_handler(request: Request, exc: EntityNotFoundError):
        return JSONResponse(
            status_code=status.HTTP_404_NOT_FOUND,
            content={"detail": exc.message},
        )

    @app.exception_handler(PermissionDeniedError)
    async def permission_denied_handler(request: Request, exc: PermissionDeniedError):
        return JSONResponse(
            status_code=status.HTTP_403_FORBIDDEN,
            content={"detail": exc.message},
        )

    @app.exception_handler(TeamIsFullError)
    async def team_is_full_handler(request: Request, exc: TeamIsFullError):
        return JSONResponse(
            status_code=status.HTTP_409_CONFLICT,
            content={"detail": exc.message},
        )

    @app.exception_handler(AlreadyInTeamError)
    async def already_in_team_handler(request: Request, exc: AlreadyInTeamError):
        return JSONResponse(
            status_code=status.HTTP_409_CONFLICT,
            content={"detail": exc.message},
        )

    @app.exception_handler(DeadlinePassedError)
    async def deadline_passed_handler(request: Request, exc: DeadlinePassedError):
        return JSONResponse(
            status_code=status.HTTP_403_FORBIDDEN,
            content={"detail": exc.message},
        )

    @app.exception_handler(IncompleteTeamError)
    async def incomplete_team_handler(request: Request, exc: IncompleteTeamError):
        return JSONResponse(
            status_code=status.HTTP_400_BAD_REQUEST,
            content={"detail": exc.message},
        )

    @app.exception_handler(InvalidStateError)
    async def invalid_state_handler(request: Request, exc: InvalidStateError):
        return JSONResponse(
            status_code=status.HTTP_400_BAD_REQUEST,
            content={"detail": exc.message},
        )

    @app.exception_handler(DomainError)
    async def domain_error_handler(request: Request, exc: DomainError):
        return JSONResponse(
            status_code=status.HTTP_400_BAD_REQUEST,
            content={"detail": exc.message},
        )

import uuid
from typing import Optional

from fastapi import Depends, Request
from fastapi_users import BaseUserManager, UUIDIDMixin, exceptions, models, schemas

from auth.database import get_user_db
from config import SECRET_USER_MANAGER
from models.user import User

SECRET = SECRET_USER_MANAGER


class UserManager(UUIDIDMixin, BaseUserManager[User, uuid.UUID]):
    """Менеджер жизненного цикла пользователя (регистрация, сброс пароля, верификация)."""

    reset_password_token_secret = SECRET
    verification_token_secret = SECRET

    async def on_after_register(self, user: User, request: Optional[Request] = None):
        """Хэндлер после успешной регистрации пользователя."""
        print(f"Пользователь {user.id} ({user.email}) успешно зарегистрирован.")

    async def on_after_forgot_password(
        self, user: User, token: str, request: Optional[Request] = None
    ):
        """Хэндлер запроса сброса пароля."""
        print(f"Пользователь {user.id} запросил сброс пароля. Токен: {token}")

    async def on_after_request_verify(
        self, user: User, token: str, request: Optional[Request] = None
    ):
        """Хэндлер запроса верификации email."""
        print(f"Запрошена верификация для пользователя {user.id}. Токен: {token}")

    async def create(
        self,
        user_create: schemas.UC,
        safe: bool = False,
        request: Optional[Request] = None,
    ) -> models.UP:
        """Создание пользователя с валидацией пароля и хэшированием."""
        await self.validate_password(user_create.password, user_create)

        existing_user = await self.user_db.get_by_email(user_create.email)
        if existing_user is not None:
            raise exceptions.UserAlreadyExists()

        user_dict = (
            user_create.create_update_dict()
            if safe
            else user_create.create_update_dict_superuser()
        )
        password = user_dict.pop("password")
        user_dict["hashed_password"] = self.password_helper.hash(password)

        created_user = await self.user_db.create(user_dict)
        await self.on_after_register(created_user, request)
        return created_user


async def get_user_manager(user_db=Depends(get_user_db)):
    """Фабрика зависимости менеджера пользователей."""
    yield UserManager(user_db)

"""CLI-команда для создания первого администратора/организатора платформы."""

import argparse
import asyncio
import sys

from sqlalchemy import select

from auth.auth import password_helper
from database.database import async_session_maker
from models.enums import UserRole
from models.user import User


async def create_superuser(
    email: str,
    password: str,
    full_name: str,
    role: UserRole = UserRole.ADMIN,
) -> None:
    """Создает супер-администратора или организатора в базе данных."""
    async with async_session_maker() as session:
        # Проверяем, существует ли уже пользователь с таким email
        stmt = select(User).where(User.email == email)
        result = await session.execute(stmt)
        existing_user = result.scalar_one_or_none()

        if existing_user:
            print(f"Ошибка: Пользователь с email '{email}' уже существует.")
            sys.exit(1)

        hashed_password = password_helper.hash(password)
        new_user = User(
            email=email,
            hashed_password=hashed_password,
            full_name=full_name,
            role=role,
            is_active=True,
            is_superuser=True,
            is_verified=True,
            skills=[],
            specializations=[],
        )
        session.add(new_user)
        await session.commit()
        await session.refresh(new_user)
        print(
            f"Супер-администратор успешно создан: ID={new_user.id}, Email={new_user.email}, Роль={new_user.role.value}"
        )


def main() -> None:
    """Точка входа CLI."""
    parser = argparse.ArgumentParser(
        description="Создание первого администратора платформы TeamFinder"
    )
    parser.add_argument(
        "--email",
        type=str,
        required=True,
        help="Email адрес администратора",
    )
    parser.add_argument(
        "--password",
        type=str,
        required=True,
        help="Пароль администратора (минимум 8 символов)",
    )
    parser.add_argument(
        "--full-name",
        type=str,
        default="Главный Администратор",
        help="ФИО администратора",
    )
    parser.add_argument(
        "--role",
        type=str,
        choices=[r.value for r in UserRole],
        default=UserRole.ADMIN.value,

def run_migrations_offline() -> None:
    """Запуск миграций в offline-режиме."""
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )

    with context.begin_transaction():
        context.run_migrations()


def do_run_migrations(connection: Connection) -> None:
    """Синхронный колбэк, который Alembic выполняет внутри асинхронного соединения."""
    context.configure(connection=connection, target_metadata=target_metadata)

    with context.begin_transaction():
        context.run_migrations()


async def run_async_migrations() -> None:
    """Создает асинхронный движок на базе asyncpg и запускает миграции."""
    connectable = async_engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    async with connectable.connect() as connection:
        await connection.run_sync(do_run_migrations)

    await connectable.dispose()


def run_migrations_online() -> None:
    """Запуск миграций в online-режиме через цикл событий asyncio."""
    asyncio.run(run_async_migrations())


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

import uuid
from typing import List, Optional, Union

from sqlalchemy import func, select

from models.enums import EventStatus
from models.event import Event
from utils.repository import SQLAlchemyRepository


class EventRepository(SQLAlchemyRepository):
    """Репозиторий таблицы мероприятий."""

    model = Event

    async def get_by_id_or_invite_code(
        self,
        identifier: Union[str, uuid.UUID],
    ) -> Optional[Event]:
        """Поиск мероприятия по UUID или по инвайт-коду.

        Сначала пытается обработать идентификатор как UUID через базовый find_one_or_none(id=...),
        а если строка не является UUID или запись не найдена — ищет по invite_code.
        """
        if isinstance(identifier, uuid.UUID):
            return await self.find_one_or_none(id=identifier)

        try:
            parsed_uuid = uuid.UUID(identifier)
            event = await self.find_one_or_none(id=parsed_uuid)
            if event is not None:
                return event
        except (ValueError, AttributeError):
            pass

        return await self.find_one_or_none(invite_code=str(identifier))

    async def get_public_events(
        self,
        is_featured: Optional[bool] = None,
        status: Optional[EventStatus] = None,
        limit: int = 100,
        offset: int = 0,
    ) -> List[Event]:
        """Получение списка публичных мероприятий для главной страницы с сортировкой и пагинацией.

        :param is_featured: фильтр для карточек верхнего слайдера
        :param status: фильтр по статусу жизненного цикла (например, REGISTRATION)
        :param limit: количество записей (пагинация)
        :param offset: смещение (пагинация)
        """
        stmt = select(self.model).where(self.model.is_public.is_(True))

        if is_featured is not None:
            stmt = stmt.where(self.model.is_featured == is_featured)

        if status is not None:
            stmt = stmt.where(self.model.status == status)

        stmt = stmt.order_by(self.model.event_start.asc()).limit(limit).offset(offset)
        res = await self.session.execute(stmt)
        return list(res.scalars().all())

    async def count_public_events(
        self,
        is_featured: Optional[bool] = None,
        status: Optional[EventStatus] = None,
    ) -> int:
        """Подсчет общего количества публичных мероприятий с учетом фильтров."""
        stmt = select(func.count(self.model.id)).where(self.model.is_public.is_(True))

        if is_featured is not None:
            stmt = stmt.where(self.model.is_featured == is_featured)

        if status is not None:
            stmt = stmt.where(self.model.status == status)

        res = await self.session.execute(stmt)
        return res.scalar_one() or 0

"""Репозиторий для работы с регистрациями участников на мероприятия (EventRegistration)."""

import uuid
from typing import List, Optional

from sqlalchemy import func, or_, select
from sqlalchemy.orm import joinedload

from models.event import EventRegistration
from models.user import User
from utils.repository import SQLAlchemyRepository


class EventRegistrationRepository(SQLAlchemyRepository):
    """Репозиторий таблицы регистраций участников на мероприятия."""

    model = EventRegistration

    async def get_solo_participants(
        self,
        event_id: uuid.UUID,
        tags: Optional[List[str]] = None,
        limit: int = 100,
        offset: int = 0,
    ) -> List[EventRegistration]:
        """Получить свободных участников мероприятия без команды (team_id IS NULL).

        Выполняет JOIN с пользователями, жадную предзагрузку связанных данных профиля (joinedload)
        и фильтрацию по массиву тегов пользователя (User.skills.overlap).

        :param event_id: ID мероприятия
        :param tags: список тегов для проверки пересечения с навыками и компетенциями
        :param limit: лимит записей (пагинация)
        :param offset: смещение (пагинация)
        """
        stmt = (
            select(self.model)
            .options(joinedload(self.model.user))
            .join(User, self.model.user_id == User.id)
            .where(
                self.model.event_id == event_id,
                self.model.team_id.is_(None),
            )
        )

        if tags:
            stmt = stmt.where(
                or_(
                    User.specializations.overlap(tags),
                    User.skills.overlap(tags),
                )
            )

        stmt = stmt.order_by(self.model.registered_at.asc()).limit(limit).offset(offset)
        res = await self.session.execute(stmt)
        return list(res.scalars().all())

    async def count_solo_participants(
        self,
        event_id: uuid.UUID,
        tags: Optional[List[str]] = None,
    ) -> int:
        """Подсчет общего количества свободных участников с учетом тегов."""
        stmt = (
            select(func.count(self.model.id))
            .join(User, self.model.user_id == User.id)
            .where(
                self.model.event_id == event_id,
                self.model.team_id.is_(None),
            )
        )

        if tags:
            stmt = stmt.where(
                or_(
                    User.specializations.overlap(tags),
                    User.skills.overlap(tags),
                )
            )

        res = await self.session.execute(stmt)
        return res.scalar_one() or 0

    async def get_by_event_and_user_for_update(
        self,
        event_id: uuid.UUID,
        user_id: uuid.UUID,
    ) -> Optional[EventRegistration]:
        """Найти регистрацию с пессимистичной блокировкой (SELECT FOR UPDATE) для защиты от Race Condition."""
        stmt = (
            select(self.model)
            .where(
                self.model.event_id == event_id,
                self.model.user_id == user_id,
            )
            .with_for_update()
        )
        res = await self.session.execute(stmt)
        return res.scalar_one_or_none()

    async def get_by_event_and_user(
        self,
        event_id: uuid.UUID,
        user_id: uuid.UUID,
    ) -> Optional[EventRegistration]:

        stmt = (

            )
            .where(
                self.model.event_id == event_id,
                self.model.user_id == user_id,
            )
        )
        res = await self.session.execute(stmt)
        return res.scalar_one_or_none()

    async def get_user_registrations(

    ) -> List[EventRegistration]:
        """Получить все регистрации пользователя с предзагрузкой мероприятий и команд (для профиля)."""
        stmt = (
            select(self.model)
            .options(

                joinedload(self.model.team),
            )
            .where(self.model.user_id == user_id)
            .order_by(self.model.registered_at.desc())
        )
        res = await self.session.execute(stmt)
        return list(res.scalars().all())

            TeamRequestStatus.PENDING_USER,
        ]
        # Сначала получаем заявки, чтобы иметь возможность отправить уведомления
        stmt_select = select(self.model).where(
            self.model.team_id == team_id,
            self.model.status.in_(active_statuses),
        )
        if exclude_request_id is not None:
            stmt_select = stmt_select.where(self.model.id != exclude_request_id)

        res = await self.session.execute(stmt_select)
        requests_to_cancel = list(res.scalars().all())

        if requests_to_cancel:
            cancel_ids = [r.id for r in requests_to_cancel]
            stmt_update = (
                update(self.model)
                .where(self.model.id.in_(cancel_ids))
                .values(status=TeamRequestStatus.CANCELLED)
            )
            await self.session.execute(stmt_update)

        return requests_to_cancel

    async def cancel_all_event_pending_requests(self, event_id: uuid.UUID) -> None:
        """Массово отменить все активные заявки хакатона при наступлении TEAM_LOCK."""
        active_statuses = [
            TeamRequestStatus.PENDING_CAPTAIN,
            TeamRequestStatus.PENDING_USER,
        ]
        stmt = (
            update(self.model)
            .where(
                self.model.event_id == event_id,
                self.model.status.in_(active_statuses),
            )
            .values(status=TeamRequestStatus.CANCELLED)
        )
        await self.session.execute(stmt)

    async def get_user_event_requests(
        self,
        event_id: uuid.UUID,
        user_id: uuid.UUID,
    ) -> List[TeamRequest]:
        """Получить все заявки пользователя в рамках мероприятия."""
        stmt = (
            select(self.model)
            .where(
                self.model.event_id == event_id,
                self.model.user_id == user_id,
            )
            .order_by(self.model.created_at.desc())
        )

        return list(res.scalars().all())

import uuid

from sqlalchemy import func, select
from sqlalchemy.orm import joinedload, selectinload

from models.event import EventRegistration
from models.team import Team
from utils.repository import SQLAlchemyRepository


class TeamRepository(SQLAlchemyRepository):
    """Репозиторий таблицы команд."""

    model = Team

    async def get_by_id_for_update(self, team_id: uuid.UUID) -> Optional[Team]:
        """Получить команду по ID с пессимистичной блокировкой (SELECT FOR UPDATE).

        Используется для предотвращения Race Condition при одновременном вступлении участников.
        """
        stmt = select(self.model).where(self.model.id == team_id).with_for_update()
        res = await self.session.execute(stmt)
        return res.scalar_one_or_none()

    async def get_by_id_with_members(self, team_id: uuid.UUID) -> Optional[Team]:
        """Получить команду по ID с жадной предзагрузкой капитана и участников с их профилями."""
        stmt = (
            select(self.model)
            .execution_options(populate_existing=True)
            .options(
                joinedload(self.model.captain),
                selectinload(self.model.members).joinedload(EventRegistration.user),
            )
            .where(self.model.id == team_id)
        )
        res = await self.session.execute(stmt)
        return res.scalar_one_or_none()

    def _apply_catalog_filters(
        self,
        stmt,
        needed_role: Optional[str],
        has_free_slots: Optional[bool],
    ):
        """Применить опциональные фильтры каталога команд (роль «Ищем» и свободные места)."""
        if needed_role is not None:
            stmt = stmt.where(self.model.needed_roles.any(needed_role))

        if has_free_slots is not None:
            members_count = (
                select(func.count(EventRegistration.id))
                .where(EventRegistration.team_id == self.model.id)
                .correlate(self.model)
                .scalar_subquery()
            )
            if has_free_slots:
                stmt = stmt.where(members_count < self.model.max_members)
            else:
                stmt = stmt.where(members_count >= self.model.max_members)

        return stmt

    async def get_by_event_id(
        self,
        event_id: uuid.UUID,
        needed_role: Optional[str] = None,
        has_free_slots: Optional[bool] = None,
        limit: int = 100,
        offset: int = 0,
    ) -> List[Team]:
        """Получить список всех команд мероприятия с предзагрузкой участников для каталога."""
        stmt = (
            select(self.model)
            .options(
                joinedload(self.model.captain),
                selectinload(self.model.members).joinedload(EventRegistration.user),
            )
            .where(self.model.event_id == event_id)
        )
        stmt = self._apply_catalog_filters(stmt, needed_role, has_free_slots)
        stmt = stmt.order_by(self.model.created_at.asc()).limit(limit).offset(offset)
        res = await self.session.execute(stmt)
        return list(res.scalars().all())

    async def count_by_event(
        self,
        event_id: uuid.UUID,
        needed_role: Optional[str] = None,
        has_free_slots: Optional[bool] = None,
    ) -> int:
        """Подсчёт команд мероприятия для пагинации."""
        stmt = select(func.count(self.model.id)).where(self.model.event_id == event_id)
        stmt = self._apply_catalog_filters(stmt, needed_role, has_free_slots)
        res = await self.session.execute(stmt)
        return res.scalar_one() or 0

    async def get_by_invite_token(self, invite_token: str) -> Optional[Team]:
        """Найти команду по секретному токену ссылки-приглашения (с предзагрузкой участников и события)."""
        stmt = (
            select(self.model)
            .options(
                joinedload(self.model.captain),
                joinedload(self.model.event),
                selectinload(self.model.members).joinedload(EventRegistration.user),
            )
            .where(self.model.invite_token == invite_token)
        )
        res = await self.session.execute(stmt)
        return res.scalar_one_or_none()

    async def get_by_invite_token_for_update(self, invite_token: str) -> Optional[Team]:
        """Найти команду по секретному токену с пессимистичной блокировкой (SELECT FOR UPDATE)."""
        stmt = (
            select(self.model)
            .where(self.model.invite_token == invite_token)
            .with_for_update()
        )
        res = await self.session.execute(stmt)
        return res.scalar_one_or_none()

    async def get_members_count(self, team_id: uuid.UUID) -> int:
        """Получить текущее количество подтвержденных участников в команде."""
        stmt = select(func.count(EventRegistration.id)).where(
            EventRegistration.team_id == team_id
        )
        res = await self.session.execute(stmt)
        return res.scalar_one() or 0

    async def is_name_taken(
        self,
        event_id: uuid.UUID,
        name: str,
        exclude_team_id: Optional[uuid.UUID] = None,
    ) -> bool:
        """Проверить, занято ли название команды в рамках мероприятия (без учета регистра)."""
        clean_name = name.strip().lower()
        stmt = select(func.count(self.model.id)).where(
            self.model.event_id == event_id,
            func.lower(self.model.name) == clean_name,
        )
        if exclude_team_id is not None:
            stmt = stmt.where(self.model.id != exclude_team_id)

        res = await self.session.execute(stmt)
        return (res.scalar_one() or 0) > 0

    async def get_incomplete_teams_for_matching(
        self, event_id: uuid.UUID
    ) -> List[Team]:
        """Получить неполные команды хакатона с флагом allow_auto_match=True для умного матчинга."""
        stmt = (
            select(self.model)
            .options(
                selectinload(self.model.members).joinedload(EventRegistration.user)
            )
            .where(
                self.model.event_id == event_id,
                self.model.allow_auto_match.is_(True),
            )
            .order_by(self.model.created_at.asc())
        )
        res = await self.session.execute(stmt)
        return list(res.scalars().all())

"""Общие Pydantic-схемы платформы."""

import math
from typing import Generic, List, TypeVar
from pydantic import BaseModel, ConfigDict, Field

T = TypeVar("T")


class PaginatedResponse(BaseModel, Generic[T]):
    """Универсальная схема ответа для пагинированных списков."""

    items: List[T] = Field(..., description="Элементы текущей страницы")
    total: int = Field(..., ge=0, description="Общее количество записей")
    page: int = Field(..., ge=1, description="Номер текущей страницы (начиная с 1)")
    size: int = Field(..., ge=1, description="Размер страницы (лимит элементов)")
    pages: int = Field(..., ge=0, description="Общее количество страниц")

    model_config = ConfigDict(arbitrary_types_allowed=True, from_attributes=True)

    @classmethod
    def create(
        cls,
        items: List[T],
        total: int,
        page: int,
        size: int,
    ) -> "PaginatedResponse[T]":
        """Фабричный метод вычисления количества страниц и формирования схемы."""
        pages = math.ceil(total / size) if size > 0 else 0
        return cls(
            items=items,
            total=total,
            page=page,
            size=size,
            pages=pages,
        )

    def transform_team_orm(cls, data: Any) -> Any:
        """Динамически формирует список участников, вычисляет existing_roles и members_count."""
        # Если data — словарь (например, уже сериализованные данные)
        if isinstance(data, dict):
            return data

        # Если data — объект ORM модели Team
        if hasattr(data, "members") and hasattr(data, "captain_id"):
            members_list: List[Dict[str, Any]] = []
            existing_roles_seen = set()
            existing_roles_list: List[str] = []

            for reg in getattr(data, "members", []):
                user = getattr(reg, "user", None)
                if user:
                    is_captain = reg.user_id == data.captain_id
                    user_specs = getattr(user, "specializations", []) or []
                    user_skills = getattr(user, "skills", []) or []

                    members_list.append(
                        {
                            "user_id": user.id,
                            "full_name": user.full_name,
                            "nickname": user.nickname,
                            "avatar_url": user.avatar_url,
                            "is_captain": is_captain,
                            "specializations": user_specs,
                            "skills": user_skills,
                            "telegram_username": user.telegram_username,
                            "joined_team_at": reg.joined_team_at,
                        }
                    )

                    # Собираем канонические специализации участников в 'Уже есть' (строго user_specs, без skills)
                    for tag in list(user_specs):
                        if tag and tag not in existing_roles_seen:
                            existing_roles_seen.add(tag)
                            existing_roles_list.append(tag)

            return {
                "id": data.id,
                "event_id": data.event_id,
                "name": data.name,
                "description": data.description,
                "captain_id": data.captain_id,
                "max_members": data.max_members,
                "needed_roles": data.needed_roles or [],
                "existing_roles": existing_roles_list,
                "members_count": len(members_list),
                "members": members_list,
                "allow_auto_match": data.allow_auto_match,
                "created_at": data.created_at,
                "invite_token": getattr(data, "invite_token", None),
            }

        return data

class EventServices:
    """Сервис управления мероприятиями, регистрациями и каталогом участников."""

    def __init__(self, uow: IUnitOfWork) -> None:
        self.uow = uow

    # ------------------------------------------------------------------
    # Ленивая смена статусов хакатона (Lazy Transitions)
    # ------------------------------------------------------------------

    async def _apply_lazy_status_transitions(self, event: Event) -> None:
        """Автоматический перевод этапа хакатона при наступлении дедлайнов.

        1. REGISTRATION -> TEAM_LOCK (по истечении registration_end) с отменой висящих заявок.
        2. SUBMISSION -> FINISHED (по истечении submission_deadline).
        """
        now = datetime.now(timezone.utc)
        status_changed = False

        if event.status == EventStatus.REGISTRATION and now > event.registration_end:
            event.status = EventStatus.TEAM_LOCK
            await self.uow.events.update(event.id, {"status": EventStatus.TEAM_LOCK})
            # Автоматическая отмена всех незавершенных заявок и инвайтов
            await self.uow.requests.cancel_all_event_pending_requests(event.id)
            status_changed = True

        elif event.status == EventStatus.SUBMISSION and now > event.submission_deadline:
            event.status = EventStatus.FINISHED
            await self.uow.events.update(event.id, {"status": EventStatus.FINISHED})
            status_changed = True

        if status_changed:
            await self.uow.commit()

    # ------------------------------------------------------------------
    # Валидация кастомных полей анкеты
    # ------------------------------------------------------------------

    def _validate_custom_answers(
        self,
        schema: Optional[Any],
        answers: Optional[Dict[str, Any]],
    ) -> None:
        """Проверяет соответствие ответов участника схеме анкеты хакатона."""
        if not schema:
            return

        answers_dict = answers or {}
        # Схема может быть списком словарей полей
        fields_list = schema if isinstance(schema, list) else schema.get("fields", [])

        for field in fields_list:
            if not isinstance(field, dict):
                continue

            field_id = field.get("id")
            field_label = field.get("label", field_id)
            is_required = field.get("required", False)
            field_type = field.get("type", "text")
            options = field.get("options", [])

            value = answers_dict.get(field_id)

            # Проверка обязательности поля
            if is_required and (
                value is None or (isinstance(value, str) and not value.strip())
            ):
                raise InvalidStateError(
                    f"Обязательное поле анкеты '{field_label}' не заполнено"
                )

            # Проверка допустимых опций для select
            if value is not None and field_type == "select" and options:
                if value not in options:
                    raise InvalidStateError(
                        f"Недопустимое значение для поля '{field_label}'. "
                        f"Допустимые варианты: {', '.join(map(str, options))}"
                    )

    # ------------------------------------------------------------------
    # Каталог мероприятий (главная страница)
    # ------------------------------------------------------------------

    async def get_public_events(
        self,
        is_featured: Optional[bool] = None,
        status: Optional[EventStatus] = None,
        page: int = 1,
        size: int = 20,
    ) -> PaginatedResponse[Event]:
        """Получить список публичных мероприятий с пагинацией и проверкой дедлайнов."""
        offset = (page - 1) * size
        async with self.uow:
            events = await self.uow.events.get_public_events(
                is_featured=is_featured,
                status=status,
                limit=size,
                offset=offset,
            )
            # Применяем ленивые переходы для отображаемых событий
            for ev in events:
                await self._apply_lazy_status_transitions(ev)

            total = await self.uow.events.count_public_events(
                is_featured=is_featured,
                status=status,
            )
            return PaginatedResponse.create(
                items=events,
                total=total,
                page=page,
                size=size,
            )

    # ------------------------------------------------------------------
    # Детальная страница мероприятия
    # ------------------------------------------------------------------

    async def get_event_detail(
        self,
        identifier: str,
        current_user_id: Optional[uuid.UUID] = None,
    ) -> dict:
        """Получить детальную информацию о мероприятии."""
        async with self.uow:
            event = await self.uow.events.get_by_id_or_invite_code(identifier)
            if not event:
                raise EntityNotFoundError(f"Мероприятие '{identifier}' не найдено")

            await self._apply_lazy_status_transitions(event)

            my_registration = None
            if current_user_id is not None:
                my_registration = await self.uow.registrations.get_by_event_and_user(
                    event_id=event.id,
                    user_id=current_user_id,
                )

            return {
                "event": event,
                "my_registration": my_registration,
            }

    # ------------------------------------------------------------------
    # Регистрация участника на мероприятие
    # ------------------------------------------------------------------

    async def register_for_event(
        self,
        event_id: uuid.UUID,
        user_id: uuid.UUID,
        registration_data: EventRegistrationCreate,
    ) -> EventRegistration:
        """Зарегистрировать участника на мероприятие."""
        async with self.uow:
            event = await self.uow.events.find_one_or_none(id=event_id)
            if not event:
                raise EntityNotFoundError(f"Мероприятие с ID {event_id} не найдено")

            # Проверяем и применяем переход дедлайна
            await self._apply_lazy_status_transitions(event)

            if event.status != EventStatus.REGISTRATION:
                raise InvalidStateError(
                    "Регистрация на это мероприятие закрыта "
                    f"(текущий статус: {event.status.value})"
                )

            now = datetime.now(timezone.utc)
            if now > event.registration_end:
                raise InvalidStateError("Срок регистрации на мероприятие истек")

            # Валидируем кастомные ответы анкеты хакатона
            self._validate_custom_answers(
                event.custom_fields_schema,
                registration_data.custom_answers,
            )

            # Проверяем, что пользователь ещё не зарегистрирован
            existing = await self.uow.registrations.get_by_event_and_user(
                event_id=event_id,
                user_id=user_id,
            )
            if existing:
                raise InvalidStateError("Вы уже зарегистрированы на это мероприятие")

            registration = await self.uow.registrations.add_one(
                {
                    "event_id": event_id,
                    "user_id": user_id,
                    "custom_answers": registration_data.custom_answers,
                }
            )
            await self.uow.commit()
            return registration

    # ------------------------------------------------------------------
    # Каталог свободных участников (одиночки)
    # ------------------------------------------------------------------

    async def get_solo_participants(

            await self._apply_lazy_status_transitions(event)

            items = await self.uow.registrations.get_solo_participants(

                tags=tags,
                limit=size,
                offset=offset,
            )
            total = await self.uow.registrations.count_solo_participants(
                event_id=event_id,
                tags=tags,
            )
            return PaginatedResponse.create(
                items=items,
                total=total,
                page=page,
                size=size,
            )

"""Сервисный слой для работы с командами, инвайтами и составом."""

import secrets

class TeamServices:
    """Сервис управления командами и жизненным циклом состава."""

    def __init__(self, uow: IUnitOfWork) -> None:
        self.uow = uow

    def _ensure_registration_phase(self, event) -> None:
        """Проверить, что операция выполняется в фазе регистрации."""
        now = datetime.now(timezone.utc)
        if event.status != EventStatus.REGISTRATION or now > event.registration_end:
            raise InvalidStateError(
                "Операция доступна только во время фазы регистрации мероприятия"
            )

    async def _sync_event_state(self, event) -> None:
        """Применить ленивые переходы статуса мероприятия перед командной операцией."""
        await EventServices(self.uow)._apply_lazy_status_transitions(event)

    async def _generate_unique_invite_token(self) -> str:
        """Сгенерировать уникальный секретный токен приглашения."""
        while True:
            token = secrets.token_hex(32)
            existing = await self.uow.teams.find_one_or_none(invite_token=token)
            if not existing:
                return token

    @staticmethod
    def _close_one_needed_role(
        needed_roles: list[str], user_specializations: list[str]
    ) -> None:
        """Удалить не более одной закрываемой роли из needed_roles."""
        for specialization in user_specializations:
            if specialization in needed_roles:
                needed_roles.remove(specialization)
                break

    @staticmethod
    def _can_manage_team(team: Team, event: Any, actor_user: User) -> bool:
        """Проверить право управления командой."""
        return bool(
            actor_user.is_superuser
            or team.captain_id == actor_user.id
            or event.organizer_id == actor_user.id
        )

    @staticmethod
    def _build_team_detail_payload(
        team: Team,
        event: Any,
        current_user: Optional[User],
    ) -> dict:
        """Собрать безопасный payload для TeamDetailRead."""
        team_payload = TeamDetailRead.model_validate(team).model_dump()

        is_user_member = bool(
            current_user
            and any(
                member.user_id == current_user.id for member in (team.members or [])
            )
        )
        is_user_captain = bool(current_user and team.captain_id == current_user.id)
        can_view_invite_token = bool(
            current_user
            and (
                is_user_member
                or event.organizer_id == current_user.id
                or current_user.is_superuser
            )
        )

        team_payload.update(
            {
                "invite_token": team.invite_token if can_view_invite_token else None,
                "is_user_member": is_user_member,
                "is_user_captain": is_user_captain,
            }
        )
        return team_payload

    async def create_team(
        self,
        event_id: uuid.UUID,
        creator_id: uuid.UUID,
        team_data: TeamCreate,
    ) -> Team:
        """Создать команду и автоматически зачислить создателя как капитана."""
        async with self.uow:
            event = await self.uow.events.find_one_or_none(id=event_id)
            if not event:
                raise EntityNotFoundError(f"Мероприятие с ID {event_id} не найдено")
            await self._sync_event_state(event)
            self._ensure_registration_phase(event)

            registration = (
                await self.uow.registrations.get_by_event_and_user_for_update(
                    event_id=event_id,
                    user_id=creator_id,
                )
            )
            if not registration:
                raise InvalidStateError(
                    "Чтобы создать команду, нужно сначала зарегистрироваться на мероприятие"
                )
            if registration.team_id is not None:
                raise AlreadyInTeamError()

            if await self.uow.teams.is_name_taken(event_id, team_data.name):
                raise InvalidStateError(
                    "Команда с таким названием уже существует на этом мероприятии"
                )

            if team_data.max_members < event.min_team_size:
                raise InvalidStateError(
                    "Желаемый размер команды не может быть меньше минимального размера команды на мероприятии"
                )
            if team_data.max_members > event.max_team_size:
                raise InvalidStateError(
                    "Желаемый размер команды не может превышать лимит мероприятия"
                )

            invite_token = await self._generate_unique_invite_token()
            team_payload = team_data.model_dump(mode="json")
            team_payload.update(
                {
                    "event_id": event_id,
                    "captain_id": creator_id,
                    "invite_token": invite_token,
                }
            )
            team = await self.uow.teams.add_one(team_payload)

            now = datetime.now(timezone.utc)
            await self.uow.registrations.update(
                registration.id,
                {
                    "team_id": team.id,
                    "joined_team_at": now,
                },
            )
            await self.uow.requests.cancel_other_user_requests(event_id, creator_id)
            await self.uow.commit()

            return await self.uow.teams.get_by_id_with_members(team.id)

            event = await self.uow.events.find_one_or_none(id=team.event_id)
            if not event:
                raise EntityNotFoundError(

            if not self._can_manage_team(team, event, actor_user):

            updated_team = await self.uow.teams.get_by_id_with_members(team_id)
            if not updated_team:

            return updated_team

    async def regenerate_invite_token(

                    },
                )

            await self.uow.teams.update(team.id, {"needed_roles": needed_roles})

            await self.uow.requests.cancel_other_user_requests(event.id, user_id)

            if current_members_count + 1 >= limit:
                await self.uow.requests.cancel_team_pending_requests(team.id)

import hashlib
import secrets
from pathlib import Path
from typing import Set

from fastapi import UploadFile
from PIL import Image

from utils.exceptions import InvalidStateError

# Разрешенные MIME-типы и расширения
ALLOWED_IMAGE_TYPES: Set[str] = {"image/jpeg", "image/png", "image/webp"}
MAX_IMAGE_SIZE_BYTES = 5 * 1024 * 1024  # 5 Мегабайт

# Директория загрузки статики: backend/static/uploads/
BASE_DIR = Path(__file__).resolve().parent.parent.parent
UPLOAD_DIR = BASE_DIR / "static" / "uploads"


class UploadServices:
    """Сервис обработки медиа-файлов."""

    def __init__(self, upload_dir: Path = UPLOAD_DIR):
        self.upload_dir = upload_dir
        self.upload_dir.mkdir(parents=True, exist_ok=True)

    async def save_image(self, file: UploadFile) -> str:
        """Валидирует и сохраняет загруженное изображение.

        Возвращает относительный URL для доступа к файлу.
        """
        # 1. Проверка Content-Type
        if file.content_type not in ALLOWED_IMAGE_TYPES:
            raise InvalidStateError(
                f"Неподдерживаемый формат файла '{file.content_type}'. "
                "Разрешены только JPEG, PNG и WebP."
            )

        # 2. Чтение содержимого с проверкой размера
        contents = await file.read()
        if len(contents) > MAX_IMAGE_SIZE_BYTES:
            raise InvalidStateError(
                f"Превышен допустимый размер файла ({len(contents) / (1024 * 1024):.1f} МБ). "
                "Максимальный размер: 5 МБ."
            )

        # 3. Валидация целостности картинки через Pillow
        try:
            import io

            image = Image.open(io.BytesIO(contents))
            image.verify()
        except Exception:
            raise InvalidStateError(
                "Файл поврежден или не является валидным изображением"
            )

        # 4. Формирование уникального имени файла
        ext = (
            file.filename.split(".")[-1].lower()
            if file.filename and "." in file.filename
            else "webp"
        )
        if ext not in ("jpg", "jpeg", "png", "webp"):
            ext = "webp"

        file_hash = hashlib.sha256(contents).hexdigest()[:16]
        unique_name = f"{file_hash}_{secrets.token_hex(4)}.{ext}"
        destination_path = self.upload_dir / unique_name

        # 5. Запись на диск
        with open(destination_path, "wb") as f:
            f.write(contents)

        return f"/static/uploads/{unique_name}"

        self,
        user_id: uuid.UUID,

            if not user:

            hashed_password = password_helper.hash(password_data.new_password)
            await self.uow.users.update(user_id, {"hashed_password": hashed_password})
            await self.uow.commit()

                )

                "telegram_id": telegram_id,
                "telegram_username": telegram_username,
                "is_bot_active": True,
                "telegram_bind_token": None,
                "telegram_bind_token_expires_at": None,
            }
            updated_user = await self.uow.users.update(user.id, update_data)
            await self.uow.commit()
            return updated_user

from abc import ABC, abstractmethod
from typing import Any
from sqlalchemy import insert, select, delete, update
from sqlalchemy.ext.asyncio import AsyncSession


class AbstractRepository(ABC):
    @abstractmethod
    async def add_one(self, data: dict) -> Any:
        raise NotImplementedError

    @abstractmethod
    async def find_all(self, **filter_by) -> list[Any]:
        raise NotImplementedError

    @abstractmethod
    async def find_one_or_none(self, **filter_by) -> Any | None:
        raise NotImplementedError

    @abstractmethod
    async def find_filter(self, *filters) -> list[Any]:
        raise NotImplementedError

    @abstractmethod
    async def update(self, id: Any, values: dict) -> Any:
        raise NotImplementedError

    @abstractmethod
    async def delete_one(self, id: Any) -> Any:
        raise NotImplementedError


class SQLAlchemyRepository(AbstractRepository):
    model = None

    def __init__(self, session: AsyncSession):
        # Используем ТОЛЬКО эту общую сессию от UnitOfWork во всех методах!
        self.session = session

    async def add_one(self, data: dict):
        stmt = insert(self.model).values(**data).returning(self.model)
        res = await self.session.execute(stmt)
        return res.scalar_one()

    async def find_all(self, **filter_by):
        # Работает и без аргументов find_all(), и с именными фильтрами find_all(course=1)
        stmt = select(self.model).filter_by(**filter_by)
        res = await self.session.execute(stmt)
        return list(res.scalars().all())

    async def find_one_or_none(self, **filter_by):
        stmt = select(self.model).filter_by(**filter_by)
        res = await self.session.execute(stmt)
        return res.scalar_one_or_none()

    async def find_filter(self, *filters):
        # Для сложных условий (>, <, in_): find_filter(User.course >= 2)
        stmt = select(self.model).where(*filters)
        res = await self.session.execute(stmt)
        return list(res.scalars().all())

    async def update(self, id: Any, values: dict):
        stmt = (
            update(self.model)
            .where(self.model.id == id)
            .values(**values)
            .returning(self.model)
        )
        res = await self.session.execute(stmt)
        return res.scalar_one_or_none()

    async def delete_one(self, id: Any):
        stmt = delete(self.model).where(self.model.id == id).returning(self.model.id)
        res = await self.session.execute(stmt)
        return res.scalar_one_or_none()

"""Паттерн Unit of Work для управления транзакциями базы данных."""

from abc import ABC, abstractmethod

from database.database import async_session_maker
from repositories.event_repo import EventRepository
from repositories.registration_repo import EventRegistrationRepository
from repositories.request_repo import RequestRepository
from repositories.submission_repo import SubmissionRepository
from repositories.team_repo import TeamRepository
from repositories.user_repo import UserRepository


class IUnitOfWork(ABC):
    """Интерфейс единицы работы (Unit of Work)."""

    users: UserRepository
    events: EventRepository
    registrations: EventRegistrationRepository
    teams: TeamRepository
    requests: RequestRepository
    submissions: SubmissionRepository

    @abstractmethod
    def __init__(self): ...

    @abstractmethod
    async def __aenter__(self): ...

    @abstractmethod
    async def __aexit__(self, *args): ...

    @abstractmethod
    async def commit(self): ...

    @abstractmethod
    async def rollback(self): ...


class UnitOfWork(IUnitOfWork):
    """Реализация Unit of Work на базе SQLAlchemy AsyncSession."""

    def __init__(self):
        self.session_factory = async_session_maker

    async def __aenter__(self):
        # При входе в контекстный менеджер создается единая сессия для всех репозиториев
        self.session = self.session_factory()
        self.users = UserRepository(self.session)
        self.events = EventRepository(self.session)
        self.registrations = EventRegistrationRepository(self.session)
        self.teams = TeamRepository(self.session)
        self.requests = RequestRepository(self.session)
        self.submissions = SubmissionRepository(self.session)
        return self

    async def __aexit__(self, exc_type, exc_val, exc_tb):
        if exc_type is not None:
            await self.rollback()
        await self.session.close()

    async def commit(self):
        """Фиксация всех изменений транзакции."""
        await self.session.commit()

    async def rollback(self):
        """Откат транзакции при возникновении ошибки."""
        await self.session.rollback()
